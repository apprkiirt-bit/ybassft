const values = {};
for (const tab of tabs) for (const card of tab.cards) {
  const base = values[card.path] = { enabled: !!card.toggle && !!card.value };
  for (const f of card.fields || []) {
    if (!f.id) continue;
    if (f.type === "select") base[f.id] = f.value || (f.options || [])[0];
    else if (f.type === "slider") base[f.id] = f.value !== undefined ? f.value : f.min;
    else if (f.type === "checkbox") base[f.id] = !!f.value;
    else if (f.type === "multi") base[f.id] = {};
    else if (f.type === "keybind") base[f.id] = f.value || null;
  }
}

const el = (tag, cls, html) => { const n = document.createElement(tag); if (cls) n.className = cls; if (html !== undefined) n.innerHTML = html; return n; };
const img = (name, cls) => { const n = el("img", cls); n.src = ICON(name); return n; };
const clear = n => { while (n.firstChild) n.removeChild(n.firstChild); };

function toast(title, text) {
  const box = el("div", "toast");
  const head = el("div", "tt"); head.appendChild(img("bell"));
  const t = el("span"); t.textContent = title; head.appendChild(t); box.appendChild(head);
  if (text) { const b = el("div", "tb"); b.textContent = text; box.appendChild(b); }
  document.getElementById("toasts").appendChild(box);
  setTimeout(() => { box.style.transition = "opacity .25s,transform .25s"; box.style.opacity = "0"; box.style.transform = "translateX(10px)"; setTimeout(() => box.remove(), 260); }, 2600);
}

function closeMenus() { clear(overlay); }
function openMenu(anchor, options, current, onPick, multi) {
  closeMenus();
  const box = el("div", "menu");
  const rect = anchor.getBoundingClientRect();
  const wrect = win.getBoundingClientRect();
  box.style.left = (rect.left - wrect.left) + "px";
  box.style.top = (rect.top - wrect.top + rect.height + 4) + "px";
  box.style.width = rect.width + "px";
  for (const option of options) {
    const row = el("div", "opt" + (!multi && option === current ? " sel" : ""));
    if (multi) {
      const on = !!values[multi.card][multi.field][option];
      const bx = el("div", "bx" + (on ? " on" : "")); bx.appendChild(img("check"));
      row.appendChild(bx);
      row.appendChild(document.createTextNode(option));
      row.onclick = () => {
        values[multi.card][multi.field][option] = !values[multi.card][multi.field][option];
        bx.classList.toggle("on");
        multi.onChange && multi.onChange();
      };
    } else {
      row.appendChild(img("chevron-down"));
      row.appendChild(document.createTextNode(option));
      row.onclick = () => { onPick(option); closeMenus(); };
    }
    box.appendChild(row);
  }
  overlay.appendChild(box);
}
document.addEventListener("click", e => { if (!e.target.closest(".menu") && !e.target.closest(".sel") && !e.target.closest(".themepop")) closeMenus(); });

function select(name, onDone) {
  const icon = name.icon || "list-filter";
  return { icon, onDone };
}

function buildCard(card) {
  const box = el("div", "card");
  const st = values[card.path];
  const head = el("div", "head");
  head.appendChild(img(card.icon || "box", "ico"));
  const title = el("div", "title"); title.textContent = card.title; head.appendChild(title);
  head.appendChild(img("star", "star"));
  if (card.pill) { const p = el("div", "pill"); p.textContent = card.pill; head.appendChild(p); }
  if (card.toggle) {
    const tg = el("div", "tg" + (st.enabled ? " on" : "")); tg.appendChild(el("b"));
    tg.onclick = () => {
      st.enabled = !st.enabled; tg.classList.toggle("on");
      if (st.enabled) toast(card.title, "enabled"); else toast(card.title, "disabled");
    };
    head.appendChild(tg);
  }
  box.appendChild(head);
  if (card.desc) { const d = el("div", "desc"); d.textContent = card.desc; box.appendChild(d); }
  if (card.fields && card.fields.length) box.appendChild(el("div", "sep"));

  for (const f of card.fields || []) {
    if (f.type === "select") {
      const wrap = el("div", "field");
      const label = el("div", "lab"); label.textContent = f.label || ""; wrap.appendChild(label);
      const sel = el("div", "sel");
      sel.appendChild(img(f.icon || "list-filter", "i"));
      const span = el("span"); span.textContent = st[f.id]; sel.appendChild(span);
      sel.appendChild(img("chevron-down", "chev"));
      sel.onclick = e => {
        e.stopPropagation();
        openMenu(sel, f.options, st[f.id], option => { st[f.id] = option; span.textContent = option; toast((f.label || card.title), option); });
      };
      wrap.appendChild(sel); box.appendChild(wrap);
    }
    else if (f.type === "multi") {
      const wrap = el("div", "field");
      const label = el("div", "lab"); label.textContent = f.label || ""; wrap.appendChild(label);
      const sel = el("div", "sel");
      sel.appendChild(img(f.icon || "layers", "i"));
      const span = el("span");
      const refresh = () => { const n = Object.values(st[f.id]).filter(Boolean).length; span.textContent = n ? n + " selected" : "None selected"; };
      refresh();
      sel.appendChild(span);
      sel.appendChild(img("chevron-down", "chev"));
      sel.onclick = e => {
        e.stopPropagation();
        openMenu(sel, f.options, null, null, { card: card.path, field: f.id, onChange: refresh });
      };
      wrap.appendChild(sel); box.appendChild(wrap);
    }
    else if (f.type === "slider") {
      const row = el("div", "row");
      row.appendChild(el("span", null, f.label || ""));
      const val = el("span", "val"); row.appendChild(val); box.appendChild(row);
      const range = el("div", "range");
      const fill = el("u"), knob = el("s");
      range.appendChild(el("i")); range.appendChild(fill); range.appendChild(knob);
      const dec = f.decimals || 0, suf = f.suffix || "";
      const render = () => {
        const a = (st[f.id] - f.min) / (f.max - f.min);
        fill.style.width = (a * 100) + "%";
        knob.style.left = (a * 100) + "%";
        val.textContent = st[f.id].toFixed(dec) + suf;
      };
      render();
      const move = e => {
        const r = range.getBoundingClientRect();
        let a = Math.min(1, Math.max(0, (e.clientX - r.left) / r.width));
        let v = f.min + (f.max - f.min) * a;
        v = dec ? Math.round(v * Math.pow(10, dec)) / Math.pow(10, dec) : Math.round(v);
        st[f.id] = v; render();
      };
      range.onpointerdown = e => { range.setPointerCapture(e.pointerId); move(e); range.onpointermove = move; };
      range.onpointerup = () => { range.onpointermove = null; };
      box.appendChild(range);
    }
    else if (f.type === "checkbox") {
      const row = el("div", "chk");
      const left = el("div");
      const l = el("div", "l"); l.textContent = f.label || ""; left.appendChild(l);
      if (f.hint) { const s = el("div", "s"); s.textContent = f.hint; left.appendChild(s); }
      row.appendChild(left);
      const bx = el("div", "box" + (st[f.id] ? " on" : "")); bx.appendChild(img("check")); row.appendChild(bx);
      row.onclick = () => { st[f.id] = !st[f.id]; bx.classList.toggle("on"); };
      box.appendChild(row);
    }
    else if (f.type === "keybind") {
      const row = el("div", "chk");
      const l = el("div", "l"); l.textContent = f.label || ""; row.appendChild(l);
      const key = el("div", "key"); key.textContent = st[f.id] || "None";
      key.onclick = e => {
        e.stopPropagation();
        key.textContent = "Press..."; key.classList.add("listen");
        const handler = ev => {
          key.classList.remove("listen");
          if (ev.key === "Escape") { st[f.id] = null; key.textContent = "None"; }
          else { st[f.id] = ev.key; key.textContent = ev.key; }
          window.removeEventListener("keydown", handler, true);
        };
        window.addEventListener("keydown", handler, true);
      };
      row.appendChild(key); box.appendChild(row);
    }
    else if (f.type === "button") {
      const btn = el("div", "btn" + (f.role === "accent" ? " acc" : ""));
      btn.appendChild(img(f.icon || "play"));
      btn.appendChild(document.createTextNode(f.text || "Button"));
      btn.onclick = () => toast(f.text || "action", "preview only — wired in the luau stage");
      box.appendChild(btn);
    }
    else if (f.type === "divider") box.appendChild(el("div", "sep"));
    else if (f.type === "hint") { const h = el("div", "hint"); h.textContent = f.label; box.appendChild(h); }
  }
  return box;
}

function renderTab() {
  closeMenus();
  clear(body);
  const tab = tabs.find(t => t.id === state.tab);
  const cols = [el("div", "col"), el("div", "col"), el("div", "col")];
  cols.forEach(c => body.appendChild(c));
  let index = 0;
  for (const card of tab.cards) {
    const box = buildCard(card);
    const hay = (card.title + " " + (card.desc || "")).toLowerCase();
    if (state.query && !hay.includes(state.query)) box.style.display = "none";
    cols[index++ % 3].appendChild(box);
  }
  [...nav.children].forEach((b, i) => b.classList.toggle("active", tabs[i].id === state.tab));
}

function buildNav() {
  clear(nav);
  for (const tab of tabs) {
    const b = el("button"); b.title = tab.title;
    b.appendChild(img(tab.icon));
    b.onclick = () => { state.tab = tab.id; history.replaceState(null, "", "#" + tab.id); renderTab(); };
    nav.appendChild(b);
  }
}

function applyTheme(name) {
  const p = palettes[name];
  const root = document.documentElement.style;
  for (const key of Object.keys(p)) root.setProperty("--" + key, p[key]);
  document.querySelectorAll(".themepop div").forEach(d => d.classList.toggle("on", d.dataset.name === name));
}

let themePop = null;
function toggleThemes(anchor) {
  if (themePop) { themePop.remove(); themePop = null; return; }
  themePop = el("div", "themepop");
  for (const name of Object.keys(palettes)) {
    const row = el("div");
    row.dataset.name = name;
    const dot = el("i"); dot.style.background = palettes[name].accent;
    row.appendChild(dot); row.appendChild(document.createTextNode(name));
    row.onclick = () => { applyTheme(name); toast("Theme", name); };
    if (name === "Violet") row.classList.add("on");
    themePop.appendChild(row);
  }
  themePop.style.right = "38px";
  document.getElementById("win").appendChild(themePop);
}

let dragging = false, dragStart = null, resizing = false, resizeStart = null;
const topbar = document.getElementById("topbar");
topbar.addEventListener("pointerdown", e => {
  if (e.target.closest(".dot, .nav, .tools, .menu, .themepop")) return;
  dragging = true;
  const rect = win.getBoundingClientRect();
  dragStart = { x: e.clientX, y: e.clientY, left: rect.left, top: rect.top };
});
document.addEventListener("pointermove", e => {
  if (dragging && dragStart) {
    win.style.transform = "none";
    win.style.left = (dragStart.left + e.clientX - dragStart.x) + "px";
    win.style.top = (dragStart.top + e.clientY - dragStart.y) + "px";
    win.style.marginLeft = "0";
  }
});
document.addEventListener("pointerup", () => { dragging = false; });

      win.style.width = state.maximized ? (window.innerWidth - 20) + "px" : "720px";
      win.style.height = state.maximized ? (window.innerHeight - 20) + "px" : "490px";
    }
  };
});
